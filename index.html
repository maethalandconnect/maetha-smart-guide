ขอบพระคุณที่ใช้บริการ สำนักงานที่ดินจังหวัดลำปาง สาขาแม่ทะ นะคะ ท่านสามารถกดดูซ้ำตั้งแต่ขั้นตอนที่ 1 หรือกลับไปหน้าแรกเพื่อเลือกงานบริการอื่นได้ตามต้องการค่ะ

โค้ดเชื่อมฐานข้อมูล Google Apps Script (สำหรับโฮสติ้งภายนอก)

ในระบบปัจจุบัน ข้อมูลทั้งหมดถูกซิงค์ผ่านฐานข้อมูลกลางออนไลน์ให้อัตโนมัติอยู่แล้ว แต่หากเจ้าหน้าที่ต้องการนำไฟล์ index.html ไปอัปโหลดบน GitHub Pages และต้องการใช้ Google Drive / Google Script เป็นฐานข้อมูลส่วนตัวของสำนักงาน สามารถทำตามขั้นตอนนี้ได้ทันทีค่ะ:

  1. ไปที่ script.google.com แล้วกดสร้างโปรเจกต์ใหม่ (New Project)
  2. คัดลอกโค้ดด้านล่างนี้ไปวางทับทั้งหมด แล้วกด Deploy > New deployment
  3. เลือกชนิดเป็น Web app ตั้งค่า Who has access เป็น Anyone (ทุกคน)
  4. คัดลอก Web App URL ที่ได้ มาวางในช่องตั้งค่าของหน้าเจ้าหน้าที่ แล้วกดบันทึก
function doGet() {
  var prop = PropertiesService.getScriptProperties().getProperty('MAE_THA_CONFIG');
  return ContentService.createTextOutput(prop || '{}')
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var body = e.postData.contents;
  PropertiesService.getScriptProperties().setProperty('MAE_THA_CONFIG', body);
  return ContentService.createTextOutput(JSON.stringify({ status: 'ok' }))
    .setMimeType(ContentService.MimeType.JSON);
}

ยืนยันการคืนค่าเริ่มต้น?

การคืนค่าเริ่มต้นจะรีเซ็ตข้อมูลในฐานข้อมูลกลางออนไลน์กลับเป็นค่ามาตรฐานของสำนักงานที่ดินจังหวัดลำปาง สาขาแม่ทะ และซิงค์ไปยังทุกเครื่องทันที
